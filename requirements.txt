# =============================================================================
# Backend runtime dependencies (Render: `pip install -r requirements.txt`)
#
# This list is exactly what the code imports. Verified with an AST scan of
# backend/app: nothing here is unused, and nothing imported is missing.
# Removed as dead weight (zero imports anywhere in the repo): chromadb,
# sentence-transformers (pulls ~2GB torch), crewai, agno, langgraph, PyPDF2,
# soundfile, numpy, the bare `langchain` meta package. That cut the Render
# build from 10-20 minutes (and the OOM risk flagged in the old render.yaml)
# to a couple of minutes.
# =============================================================================

# Web framework & server
fastapi
uvicorn[standard]      # [standard] already provides `websockets`
python-multipart       # required for Form/File endpoints (interview setup)

# Config
pydantic
pydantic-settings
python-dotenv

# Database
sqlalchemy
psycopg2-binary        # Postgres driver for the prod DATABASE_URL (Neon/Supabase)

# LLM orchestration (only langchain-core + langchain-groq are imported)
langchain-core
langchain-groq

# Auth & security
python-jose[cryptography]
passlib[bcrypt]
# Pinned: passlib 1.7.4's bcrypt backend detection hashes a >72-byte secret,
# which bcrypt >= 4.1 rejects outright, so CryptContext(schemes=["bcrypt"]).hash()
# raises ValueError. 4.0.1 is the last release passlib works with and still
# satisfies chromadb's bcrypt>=4.0.1 floor.
bcrypt==4.0.1

# Voice: STT + TTS
openai                 # Groq Whisper is called through the OpenAI-compatible client
deepgram-sdk           # optional alternate STT provider
edge-tts               # free TTS voice
gTTS                   # fallback TTS

# Document parsing & PDF report generation
pdfplumber
reportlab              # imported at module scope by app/utils/pdf_generator.py;
                       # without it the app cannot even start

# HTTP / async file IO
httpx
requests
aiohttp
aiofiles

# Testing (CI runs backend/tests)
pytest
pytest-asyncio
